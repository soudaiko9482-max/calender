    <script type="module">
        import { initializeApp } from "https://www.gstatic.com/firebasejs/10.8.0/firebase-app.js";
        import { getAuth, createUserWithEmailAndPassword, signInWithEmailAndPassword, signOut, onAuthStateChanged } from "https://www.gstatic.com/firebasejs/10.8.0/firebase-auth.js";
        import { getFirestore, doc, getDoc, setDoc } from "https://www.gstatic.com/firebasejs/10.8.0/firebase-firestore.js";

        // あなたのFirebase設定
        const firebaseConfig = {
            apiKey: "AIzaSyAX7L_HgMpjR4BrcifVMw4b9lJkrS728_c",
            authDomain: "study-calendar-c0c94.firebaseapp.com",
            projectId: "study-calendar-c0c94",
            storageBucket: "study-calendar-c0c94.firebasestorage.app",
            messagingSenderId: "128580754964",
            appId: "1:128580754964:web:898bf3112baba571391101"
        };

        const app = initializeApp(firebaseConfig);
        const auth = getAuth(app);
        const db = getFirestore(app);

        let currentUser = null;
        let isRegisterMode = false;
        let taskList = [];
        let historyList = [];
        let subjects = ['国語', '数学', '英語', '理科', '社会', 'その他'];
        let currentDate = new Date();

        // ログイン状態の監視
        onAuthStateChanged(auth, async (user) => {
            if (user) {
                currentUser = user;
                document.getElementById('authScreen').style.display = 'none';
                document.getElementById('appScreen').style.display = 'block';
                document.getElementById('userEmailDisplay').innerText = `ログイン中: ${user.email}`;
                await loadCloudData();
                renderAll();
            } else {
                currentUser = null;
                document.getElementById('authScreen').style.display = 'block';
                document.getElementById('appScreen').style.display = 'none';
            }
        });

        // ログイン・新規登録の切り替え
        window.toggleAuthMode = function() {
            isRegisterMode = !isRegisterMode;
            document.getElementById('authTitle').innerText = isRegisterMode ? '新規アカウント登録' : 'ログイン';
            document.getElementById('authBtn').innerText = isRegisterMode ? '登録する' : 'ログイン';
            document.getElementById('authSwitchText').innerText = isRegisterMode ? 'すでにアカウントをお持ちの方はこちら' : 'アカウントをお持ちでない方は新規登録';
        }

        // 認証処理
        window.handleAuth = async function() {
            const email = document.getElementById('authEmail').value.trim();
            const password = document.getElementById('authPassword').value;

            if (!email || !password) {
                alert('メールアドレスとパスワードを入力してください！');
                return;
            }

            try {
                if (isRegisterMode) {
                    await createUserWithEmailAndPassword(auth, email, password);
                    alert('アカウント登録が完了しました！');
                } else {
                    await signInWithEmailAndPassword(auth, email, password);
                }
            } catch (error) {
                alert('エラーが発生しました: ' + error.message);
            }
        }

        window.logoutUser = async function() {
            await signOut(auth);
        }

        // クラウドからデータ取得
        async function loadCloudData() {
            if (!currentUser) return;
            try {
                const docRef = doc(db, "users", currentUser.uid);
                const docSnap = await getDoc(docRef);
                if (docSnap.exists()) {
                    const data = docSnap.data();
                    taskList = data.tasks || [];
                    historyList = data.history || [];
                    subjects = data.subjects || ['国語', '数学', '英語', '理科', '社会', 'その他'];
                } else {
                    taskList = [];
                    historyList = [];
                    subjects = ['国語', '数学', '英語', '理科', '社会', 'その他'];
                }
            } catch (e) {
                console.error("データ読み込みエラー:", e);
            }
        }

        // クラウドへデータ保存
        async function saveCloudData() {
            if (!currentUser) return;
            try {
                const docRef = doc(db, "users", currentUser.uid);
                await setDoc(docRef, {
                    tasks: taskList,
                    history: historyList,
                    subjects: subjects
                });
            } catch (e) {
                console.error("データ保存エラー:", e);
            }
        }

        window.updateSubjectDropdowns = function() {
            const subjectSelect = document.getElementById('subjectSelect');
            const filterSubject = document.getElementById('filterSubject');
            if (!subjectSelect || !filterSubject) return;

            const currentSelected = subjectSelect.value;
            const currentFilter = filterSubject.value;

            subjectSelect.innerHTML = '';
            subjects.forEach(sub => {
                const opt = document.createElement('option');
                opt.value = sub;
                opt.innerText = sub;
                subjectSelect.appendChild(opt);
            });
            if (subjects.includes(currentSelected)) subjectSelect.value = currentSelected;

            filterSubject.innerHTML = '<option value="all">すべてのジャンル</option>';
            subjects.forEach(sub => {
                const opt = document.createElement('option');
                opt.value = sub;
                opt.innerText = sub;
                filterSubject.appendChild(opt);
            });
            if (currentFilter === 'all' || subjects.includes(currentFilter)) {
                filterSubject.value = currentFilter;
            } else {
                filterSubject.value = 'all';
            }
        }

        window.addNewSubject = function() {
            const input = document.getElementById('newSubjectInput');
            if (!input) return;
            const newSub = input.value.trim();
            if (!newSub) {
                alert('追加するジャンル名を入力してください！');
                return;
            }
            if (subjects.includes(newSub)) {
                alert('そのジャンルはすでに存在します。');
                return;
            }
            subjects.push(newSub);
            saveCloudData();
            updateSubjectDropdowns();
            input.value = '';
            alert(`「${newSub}」を追加しました！`);
        }

        window.addTask = function() {
            const titleInput = document.getElementById('title');
            const typeInput = document.getElementById('type');
            const subjectSelect = document.getElementById('subjectSelect');
            const detailInput = document.getElementById('detail');
            const dueDateInput = document.getElementById('dueDate');

            if (!titleInput || !typeInput || !subjectSelect || !detailInput || !dueDateInput) return;

            const title = titleInput.value.trim();
            const type = typeInput.value;
            const subject = subjectSelect.value;
            const detail = detailInput.value.trim();
            const dueDate = dueDateInput.value;

            if (!title || !dueDate) {
                alert('タイトルと期限のどちらも入力してください！');
                return;
            }

            const newTask = {
                id: Date.now(),
                title: title,
                type: type,
                subject: subject,
                detail: detail,
                dueDate: dueDate
            };

            taskList.push(newTask);
            saveCloudData();

            const parts = dueDate.split('-');
            if (parts.length === 3) {
                currentDate = new Date(parseInt(parts[0]), parseInt(parts[1]) - 1, parseInt(parts[2]));
            }

            renderAll();
            titleInput.value = '';
            detailInput.value = '';
        }

        window.handleTaskClick = function(task) {
            const typeLabel = task.type === 'homework' ? '【宿題】' : '【小テスト】';
            const detailText = task.detail ? task.detail : '（詳細メモはありません）';
            
            const message = `${typeLabel} [${task.subject}] ${task.title}\n期限: ${task.dueDate}\n\n【詳細メモ】\n${detailText}\n\n------------------------\nこの予定を「完了」にしますか？`;
            
            if (confirm(message)) {
                confetti({ particleCount: 100, spread: 70, origin: { y: 0.6 } });

                const taskIndex = taskList.findIndex(t => t.id === task.id);
                if (taskIndex !== -1) {
                    const completedTask = taskList.splice(taskIndex, 1)[0];
                    completedTask.completedAt = new Date().toLocaleDateString('ja-JP');
                    historyList.unshift(completedTask);
                }

                saveCloudData();
                renderAll();
            }
        }

        window.clearHistory = function() {
            if (confirm('完了した履歴をすべて削除しますか？')) {
                historyList = [];
                saveCloudData();
                renderAll();
            }
        }

        window.changeMonth = function(direction) {
            currentDate.setMonth(currentDate.getMonth() + direction);
            renderCalendar();
        }

        window.renderAll = function() {
            updateSubjectDropdowns();
            renderCalendar();
            renderHistory();
        }

        window.renderCalendar = function() {
            const year = currentDate.getFullYear();
            const month = currentDate.getMonth();

            const monthYearEl = document.getElementById('monthYear');
            if (monthYearEl) {
                monthYearEl.innerText = `${year}年 ${month + 1}月`;
            }

            const selectedFilter = document.getElementById('filterSubject').value;
            const firstDayIndex = new Date(year, month, 1).getDay();
            const lastDay = new Date(year, month + 1, 0).getDate();
            const prevLastDay = new Date(year, month, 0).getDate();

            const grid = document.getElementById('daysGrid');
            if (!grid) return;
            grid.innerHTML = '';

            for (let i = firstDayIndex; i > 0; i--) {
                const dayDiv = document.createElement('div');
                dayDiv.className = 'day-cell other-month';
                dayDiv.innerHTML = `<div class="day-number">${prevLastDay - i + 1}</div>`;
                grid.appendChild(dayDiv);
            }

            for (let day = 1; day <= lastDay; day++) {
                const dayDiv = document.createElement('div');
                dayDiv.className = 'day-cell';
                
                const monthStr = String(month + 1).padStart(2, '0');
                const dayStr = String(day).padStart(2, '0');
                const dateStr = `${year}-${monthStr}-${dayStr}`;
                
                dayDiv.innerHTML = `<div class="day-number">${day}</div>`;

                const dayTasks = taskList.filter(task => {
                    const matchDate = task.dueDate === dateStr;
                    const matchSubject = (selectedFilter === 'all' || task.subject === selectedFilter);
                    return matchDate && matchSubject;
                });

                dayTasks.forEach(task => {
                    const chip = document.createElement('div');
                    chip.className = `task-chip ${task.type}`;
                    chip.innerText = `[${task.subject}] ${task.title}`;
                    chip.title = `クリックして詳細を表示: ${task.title}`;
                    chip.onclick = () => handleTaskClick(task);
                    dayDiv.appendChild(chip);
                });

                grid.appendChild(dayDiv);
            }
        }

        window.renderHistory = function() {
            const ul = document.getElementById('historyList');
            if (!ul) return;
            ul.innerHTML = '';

            if (historyList.length === 0) {
                ul.innerHTML = '<p style="color: #888; font-size: 0.85em;">まだ完了した予定はありません。</p>';
                return;
            }

            historyList.forEach(item => {
                const li = document.createElement('li');
                li.className = 'history-item';
                const typeLabel = item.type === 'homework' ? '【宿題】' : '【小テスト】';
                const detailStr = item.detail ? ` (メモ: ${item.detail})` : '';
                li.innerHTML = `
                    <div>
                        <span class="done-text">${typeLabel} [${item.subject}] ${item.title}${detailStr}</span>
                        <div style="font-size: 0.75em; color: #888;">期限: ${item.dueDate} (完了日: ${item.completedAt})</div>
                    </div>
                `;
                ul.appendChild(li);
            });
        }
    </script>
